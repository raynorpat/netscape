cmake_minimum_required(VERSION 3.20)
project(Netscape LANGUAGES C)
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake/modules")
set(browser_default ON)
if(WIN32)
 set(browser_default OFF)
endif()
option(NETSCAPE_BUILD_WINDOWS_CORE "Build experimental native Win32 core targets" ${WIN32})
option(NETSCAPE_BUILD_WINDOWS_FRONTEND "Build the native MFC Windows frontend" OFF)
option(NETSCAPE_BUILD_WINDOWS_EDITOR "Build the original Gold editor engine in the native Windows frontend" OFF)
option(NETSCAPE_BUILD_BROWSER "Build the Linux XFE browser and its libraries" ${browser_default})
option(NETSCAPE_BUILD_LEGACY "Build the legacy browser libraries" ${NETSCAPE_BUILD_BROWSER})
option(NETSCAPE_ENABLE_MOCHA "Enable the Mocha JavaScript engine" OFF)
option(NETSCAPE_BUILD_MOCHA_SHELL "Build the optional Mocha file-input shell" OFF)
option(NETSCAPE_ENABLE_SECURITY "Enable the legacy SSL/S-MIME implementation" OFF)
option(NETSCAPE_ENABLE_JAVA "Enable Java browser integration" OFF)
option(NETSCAPE_BUILD_SUN_JAVA "Build the Sun Java runtime and tools from source" OFF)
if(NETSCAPE_BUILD_WINDOWS_EDITOR)
 if(NOT WIN32 OR NOT NETSCAPE_BUILD_WINDOWS_CORE OR NOT NETSCAPE_BUILD_WINDOWS_FRONTEND)
  message(FATAL_ERROR "The Windows editor requires native Windows, NETSCAPE_BUILD_WINDOWS_CORE=ON and NETSCAPE_BUILD_WINDOWS_FRONTEND=ON")
 endif()
 enable_language(CXX)
endif()
if((NETSCAPE_BUILD_BROWSER OR NETSCAPE_BUILD_SUN_JAVA OR NETSCAPE_ENABLE_JAVA) AND NOT NETSCAPE_BUILD_LEGACY)
  message(FATAL_ERROR "The browser and Java targets require NETSCAPE_BUILD_LEGACY=ON")
endif()
include(NetscapeTargets)
include(cmake/targets/jpeg.cmake)
include(cmake/targets/nspr-headers.cmake)
include(cmake/targets/dbm.cmake)
if(NETSCAPE_BUILD_WINDOWS_CORE)
 if(NOT WIN32)
  message(FATAL_ERROR "NETSCAPE_BUILD_WINDOWS_CORE requires a native Windows toolchain")
 endif()
 if(NETSCAPE_BUILD_LEGACY OR NETSCAPE_BUILD_BROWSER OR NETSCAPE_ENABLE_JAVA OR NETSCAPE_BUILD_SUN_JAVA OR NETSCAPE_ENABLE_MOCHA OR NETSCAPE_BUILD_MOCHA_SHELL OR NETSCAPE_ENABLE_SECURITY)
  message(FATAL_ERROR "Native Windows currently builds core archives only; disable Linux XFE, Java, Mocha and security options")
 endif()
 include(cmake/targets/windows/core.cmake)
endif()
if(NETSCAPE_BUILD_WINDOWS_FRONTEND)
 if(NOT NETSCAPE_BUILD_WINDOWS_CORE)
  message(FATAL_ERROR "The Windows frontend requires NETSCAPE_BUILD_WINDOWS_CORE=ON")
 endif()
 include(cmake/targets/windows/frontend.cmake)
endif()
if(NETSCAPE_BUILD_LEGACY)
  include(NetscapePlatform)
  include(cmake/targets/libraries.cmake)
endif()
if(NETSCAPE_BUILD_BROWSER)
  include(cmake/targets/browser.cmake)
endif()

include(NetscapeInstall)
include(NetscapePackaging)
