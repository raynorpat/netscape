cmake_minimum_required(VERSION 3.20)
project(Netscape LANGUAGES C)
list(PREPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake/modules")
option(NETSCAPE_BUILD_BROWSER "Build the Linux XFE browser and its libraries" ON)
option(NETSCAPE_BUILD_LEGACY "Build the legacy browser libraries" ${NETSCAPE_BUILD_BROWSER})
option(NETSCAPE_ENABLE_MOCHA "Enable the Mocha JavaScript engine" OFF)
option(NETSCAPE_BUILD_MOCHA_SHELL "Build the optional Mocha file-input shell" OFF)
option(NETSCAPE_ENABLE_SECURITY "Enable the legacy SSL/S-MIME implementation" OFF)
option(NETSCAPE_ENABLE_JAVA "Enable Java browser integration" OFF)
option(NETSCAPE_BUILD_SUN_JAVA "Build the Sun Java runtime and tools from source" OFF)
if((NETSCAPE_BUILD_BROWSER OR NETSCAPE_BUILD_SUN_JAVA OR NETSCAPE_ENABLE_JAVA) AND NOT NETSCAPE_BUILD_LEGACY)
  message(FATAL_ERROR "The browser and Java targets require NETSCAPE_BUILD_LEGACY=ON")
endif()
include(NetscapeTargets)
include(cmake/targets/jpeg.cmake)
include(cmake/targets/nspr-headers.cmake)
include(cmake/targets/dbm.cmake)
if(NETSCAPE_BUILD_LEGACY)
  include(NetscapePlatform)
  include(cmake/targets/libraries.cmake)
endif()
if(NETSCAPE_BUILD_BROWSER)
  include(cmake/targets/browser.cmake)
endif()

include(NetscapeInstall)
include(NetscapePackaging)
